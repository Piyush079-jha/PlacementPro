function solve(N) {
  let memo = new Array(N + 1).fill(0);
  memo[0] = 1;
  for (let i = 2; i <= N; i++) {

      for (let j = Math.floor(i / 2); j > 0; j--) {
        memo[i] += memo[i - j];
      }
    }
  }

}

function solve(A) {
  let n = A.length;
  let up = new Array(n).fill(0);
  let down = new Array(n).fill(0);
  up[0] = 1;
  for (let i = 1; i < n; i++) {
    for (let j = 0; j < i; j++) {
      if (A[i] > A[j]) {
        up[i] = Math.max(up[i], up[j] + 1);
      } else if (A[i] < A[j]) {
        down[i] = Math.max(down[i], down[j] + 1);
      }
    }
    up[i] = Math.max(up[i], down[i] + 1);
    down[i] = Math.max(down[i], up[i] + 1);
  }
  return Math.max(...up) + Math.max(...down) - 1;
}

function main() {
  const readline = require('readline');

  });
}
